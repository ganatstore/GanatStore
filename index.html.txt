  <script>

    let currentLanguage =
      localStorage.getItem("ganatLanguage") || "ar";


    function updateLanguage() {

      document.documentElement.lang = currentLanguage;

      document.documentElement.dir =
        currentLanguage === "ar" ? "rtl" : "ltr";


      document.querySelectorAll("[data-ar]").forEach(element => {

        element.textContent =
          element.getAttribute("data-" + currentLanguage);

      });


      document.querySelector(".language-btn").textContent =
        currentLanguage === "ar"
          ? "English"
          : "العربية";


      /*
       * Re-apply LTR direction to phone numbers after
       * the language changes.
       */

      document.querySelectorAll(".phone-number").forEach(phone => {

        phone.style.direction = "ltr";
        phone.style.unicodeBidi = "isolate";

      });


      localStorage.setItem(
        "ganatLanguage",
        currentLanguage
      );
    }


    function toggleLanguage() {

      currentLanguage =
        currentLanguage === "ar"
          ? "en"
          : "ar";

      updateLanguage();
    }


    updateLanguage();
